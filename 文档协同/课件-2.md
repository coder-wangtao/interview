
## 一些小问题

注意，最好改变一下tsconfig.app.json的配置

```typescript
{
  "compilerOptions": {
    "verbatimModuleSyntax": false
  }
}
```

因为默认shadcn一些ui组件引入，在使用类型的时候，还是按照下面方式引入的:

```typescript
import { VariantProps } from "class-variance-authority";
```

不过， **TypeScript 新版本（>=5.0）** 默认启用了更严格的 **`importsNotUsedAsValues`** 相关行为，这会导致不再类型前面加上**type**修饰，就会报错。

所以最简单的办法，就是修改一下tsconfig中的配置。

当然，最佳实践还是大家手动将`shadcn`生成的`ui`文件夹下的相关文件，在引入类型的代码上，老老实实的加上**`type`**

```typescript
export function snow() {
  const duration = 15 * 1000;
  const animationEnd = Date.now() + duration;
  let skew = 1;

  function randomInRange(min: number, max: number) {
    return Math.random() * (max - min) + min;
  }

  (function frame() {
    const timeLeft = animationEnd - Date.now();
    const ticks = Math.max(200, 500 * (timeLeft / duration));
    skew = Math.max(0.8, skew - 0.001);

    confetti({
      particleCount: 1,
      startVelocity: 0,
      ticks: ticks,
      origin: {
        x: Math.random(),
        // since particles fall down, skew start toward the top
        y: Math.random() * skew - 0.2,
      },
      colors: ["#ffffff"],
      shapes: ["circle"],
      gravity: randomInRange(0.4, 0.6),
      scalar: randomInRange(0.4, 1),
      drift: randomInRange(-0.4, 0.4),
    });

    if (timeLeft > 0) {
      requestAnimationFrame(frame);
    }
  })();
}

export function schoolPride() {
  const end = Date.now() + 10 * 1000;

  const colors = ["#6726CB", "#E27588", "#54117F", "#FFFFFF"];

  (function frame() {
    confetti({
      particleCount: 4,
      angle: 60,
      spread: 55,
      origin: { x: 0 },
      colors: colors,
    });
    confetti({
      particleCount: 4,
      angle: 120,
      spread: 55,
      origin: { x: 1 },
      colors: colors,
    });

    if (Date.now() < end) {
      requestAnimationFrame(frame);
    }
  })();
}

export function firework() {
  const duration = 15 * 1000;
  const animationEnd = Date.now() + duration;
  const defaults = { startVelocity: 30, spread: 360, ticks: 60, zIndex: 0 };

  function randomInRange(min, max) {
    return Math.random() * (max - min) + min;
  }

  const interval = setInterval(function () {
    const timeLeft = animationEnd - Date.now();

    if (timeLeft <= 0) {
      return clearInterval(interval);
    }

    const particleCount = 50 * (timeLeft / duration);
    // 因为烟花要下落，这里要稍微设置高一点
    confetti({
      ...defaults,
      particleCount,
      origin: { x: randomInRange(0.1, 0.3), y: Math.random() - 0.2 },
    });
    confetti({
      ...defaults,
      particleCount,
      origin: { x: randomInRange(0.7, 0.9), y: Math.random() - 0.2 },
    });
  }, 250);
}

export function emoji() {
  const scalar = 2;
  const unicorn = confetti.shapeFromText({ text: randomEmoji(), scalar });

  const defaults = {
    spread: 360,
    ticks: 60,
    gravity: 0,
    decay: 0.96,
    startVelocity: 20,
    shapes: [unicorn],
    scalar,
  };

  function shoot() {
    confetti({
      ...defaults,
      particleCount: 30,
    });

    confetti({
      ...defaults,
      particleCount: 5,
      flat: true,
    });

    confetti({
      ...defaults,
      particleCount: 15,
      scalar: scalar / 2,
      shapes: ["circle"],
    });
  }

  setTimeout(shoot, 0);
  setTimeout(shoot, 100);
  setTimeout(shoot, 200);
}
```

### [robohash.org](https://robohash.org/)

**robohash.org** 是一个**免费的头像生成服务**网站，可以根据你提供的**任意字符串**（比如用户名、邮箱、随机字符等）生成一张独一无二的图片（机器人、怪物、外星人等风格）。

它的主要特点：

- **无需注册、免费使用**
- **纯前端请求，直接返回图片**（不需要 API Key）
- **生成结果是可复现的**：同一个字符串每次生成的头像都一样
- **支持不同主题**（机器人、怪物、猫、外星人等）
- **支持自定义图片尺寸**

**1. 基本使用方法**

它的基本访问格式是：

```
https://robohash.org/<你的字符串>
```

例如：

```
https://robohash.org/hello
```

会返回一张机器人风格的图片（基于 `hello` 这个字符串生成）。

**2. 常用参数**

robohash.org 通过 URL 参数控制输出，例如：

| 参数            | 说明                                                                             |
| --------------- | -------------------------------------------------------------------------------- |
| `/text`         | 输入任意字符串，生成唯一头像                                                     |
| `?size=300x300` | 设置图片尺寸（默认 300x300）                                                     |
| `?set=set1`     | 选择主题集：`set1`（机器人），`set2`（怪物），`set3`（机器人头像），`set4`（猫） |
| `?bgset=bg1`    | 设置背景（如 `bg1`, `bg2`）                                                      |
| `?gravatar=yes` | 兼容 Gravatar 的 MD5 hash 模式                                                   |

**3. 使用示例**

**生成 400×400 的猫头像**

```url
https://robohash.org/myusername?size=400x400&set=set4
```

**生成带背景的怪物头像**

```url
https://robohash.org/myid123?set=set2&bgset=bg1
```

**JavaScript 中使用**

```js
const username = "user42";
const imgUrl = `https://robohash.org/${encodeURIComponent(username)}？size=200x200&set=set1`;
document.querySelector("#avatar").src = imgUrl;
```

## page处理

为了更方便处理类型和数据验证，这里使用了`zod`库

```typescript
pnpm add zod
```

如果你没有用过zod，这里简单介绍一下：

### Zod 主要作用：

1. **定义数据结构（Schema）**
   类似于数据库字段定义，但在 JS/TS 代码里实现。
2. **运行时验证数据合法性**
   确保外部输入（如 API 请求、用户输入）是符合预期的。
3. **自动生成 TypeScript 类型**
   避免手动维护重复的 `interface` 或 `type`。

### Zod 的基本用法

```typescript
import { z } from "zod";

// 定义数据结构
const UserSchema = z.object({
  name: z.string(),
  age: z.number().min(0).max(120),
  email: z.string().email().optional(),
});

// 验证数据（成功返回数据，失败抛出错误）
const userData = UserSchema.parse({
  name: "Alice",
  age: 25,
});
// ✅ userData 的类型自动推导为 { name: string; age: number; email?: string | undefined }

// 如果数据不合法，会抛错
UserSchema.parse({
  name: "Bob",
  age: -5, // ❌ 年龄不合法
});
```

可以很方便的**从 Schema 推导 TypeScript 类型**

```typescript
type User = z.infer<typeof UserSchema>;

const user: User = {
  name: "Charlie",
  age: 30,
};
```

这样可以避免重复写：

```typescript
interface User { ... }
```

特别是我们在处理一些**`dto`转换**的时候，非常好用，比如`page.dto.ts`：

```typescript
import { z } from "zod";

/**
 * 创建Page页面的数据传输对象
 */
export const createPageSchema = z
  .object({
    emoji: z.string(),
    title: z.string(),
  })
  .required();

export type CreatePageDto = z.infer<typeof createPageSchema>;

/**
 * 删除应用的数据传输对象
 */
export const deletePageSchema = z
  .object({
    pageId: z.string(),
  })
  .required();

export type DeletePageDto = z.infer<typeof deletePageSchema>;
```

我们要进行验证，也能使用zod，可以建立管道处理

```typescript
nest g pi zod-validation
```

```typescript
import {
  ArgumentMetadata,
  Injectable,
  PipeTransform,
  BadRequestException,
} from "@nestjs/common";
import { ZodType } from "zod";

@Injectable()
export class ZodValidationPipe implements PipeTransform {
  constructor(private schema: ZodType) {}
  transform(value: any, metadata: ArgumentMetadata) {
    try {
      const parsedValue = this.schema.parse(value);
      return parsedValue;
    } catch {
      throw new BadRequestException("Validation failed");
    }
  }
}
```

## y-postgresql持久化处理

### 1. `PostgresqlPersistence.build(...)`：初始化持久化实例

这个方法创建了一个 `persistence` 对象，用于后续的读写操作

配置中包括：

- `tableName`：存储 Yjs 文档更新的表名（默认 `"yjs-writings"`）。
- `flushSize`：累计多少次更新后自动合并。
- `useIndex`，是否为 `docName` 字段建立索引以优化查询等

---

### 2. `getYDoc(docName: string)`：恢复一个 Y.Doc 实例

该方法会：

1. 查询数据库，获取此 `docName` 关联的历史更新数据；
2. 用这些更新创建并恢复一个完整的 `Y.Doc`；
3. 返回这个包含完整状态的文档对象

简单来说，就算你没有直接去操作数据库的情况下，再次调用 `ydoc.getXmlFragment(docId)` 时仍能看到内容：实际上是在内部通过 `getYDoc` 恢复并应用了之前所有存储的 updates。

---

### 3. `storeUpdate(docName: string, update: Uint8Array)`：自动持久化增量

每次 `Y.Doc` 发生改变（即产生新的 update），`y-postgresql` provider 会将这个二进制更新写入数据库：

```
ydoc.on('update', async (update: Uint8Array) => {
  pgdb.storeUpdate(docName, update);
});
```

这意味着，哪怕你没有手动操作数据库，所有内容增量都被自动捕获并存储

---

### 4. 恢复流程简述

把这些关键点整合起来，你的流程大致如下：

| 阶段                | 触发动作                           | 源码操作                                                                                                                     |
| ------------------- | ---------------------------------- | ---------------------------------------------------------------------------------------------------------------------------- |
| 应用启动 / 新建连接 | `bindState(docName, ydoc)` 被调用  | 调用 `getYDoc(docName)` → 获取历史 updates → 创建并初始化一个新 `Y.Doc` 实例；然后监听 `ydoc.on('update', ...)` 保存未来更新 |
| 文档更新            | 本地操作触发 `ydoc` 更新           | `y-postgresql` 捕获 update → `storeUpdate(docName, update)` 写入数据库。                                                     |
| 下次访问同一文档    | 再次调用 `getXmlFragment("docId")` | 实际上是在已恢复的 `Y.Doc` 上操作，所以数据都在内存里了。                                                                    |

### 5.前后端配合流程

**前端 → provider**

- 你在前端对文档做了操作（比如插入文本），`doc` 生成一个 `update`。
- `WebsocketProvider` 会通过 WebSocket 把这个 update 发给服务器。

**后端 → y-websocket server**

- 服务器端的 `y-websocket` 收到这个 update，广播给其他连接的 client。
- 同时，后端集成的 **`y-postgresql` persistence** 会调用 `storeUpdate(docName, update)`，把这个 update 存进 PostgreSQL。

**新客户端加入时**

- 新 client 连接到这个 `docName` 时，服务器会先调用 `getYDoc(docName)` 从 PostgreSQL 恢复出一个完整的 `Y.Doc`。
- 然后把它的内容同步给新 client。
- 这样你前端一旦 `provider.connect()`，就会拿到数据库里的最新内容。

**简单来说**：你可以直接在前端 getXmlFragment 就读出完整内容，是因为 **前端连上了带 `y-postgresql` 持久化的 `y-websocket` 后端**，后端已经在帮你从数据库恢复并同步最新的 Yjs 状态。
